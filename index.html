async function generateProductionKit(){ const story = document.getElementById("story").value.trim(); const status = document.getElementById("status"); const result = document.getElementById("result"); const button = document.getElementById("generate"); const actions = document.getElementById("actions"); if(!story){ status.textContent = "Please enter your story idea first."; document.getElementById("story").focus(); return; } button.disabled = true; button.textContent = "🎬 Creating your production kit..."; status.textContent = "Connecting to CinePilot AI..."; result.style.display = "none"; actions.style.display = "none"; stopReading(); const controller = new AbortController(); const timer = setTimeout(() => controller.abort(), 70000); try{ status.textContent = "Generating your screenplay and production plan..."; const response = await fetch(WORKER_URL, { method: "POST", headers: {"Content-Type": "application/json"}, body: JSON.stringify({ story, genre: document.getElementById("genre").value, mood: document.getElementById("mood").value, language: document.getElementById("language").value, duration: document.getElementById("duration").value }), signal: controller.signal }); const raw = await response.text(); let data; try { data = JSON.parse(raw); } catch { throw new Error("Worker returned an invalid response: " + raw.slice(0,250)); } if(!response.ok){ throw new Error( (data.error || "Request failed") + (data.details ? " — " + data.details : "") + (data.status ? " (HTTP " + data.status + ")" : "") ); } latestKit = data.productionKit || data.blueprint || ""; if(!latestKit){ throw new Error("Gemini returned an empty production kit."); } result.textContent = latestKit; result.style.display = "block"; actions.style.display = "grid"; status.textContent = "Your production kit is ready!"; }catch(error){ if(error.name === "AbortError"){ status.textContent = "Website timeout: request took too long. Try a shorter story."; }else{ status.textContent = "Error: " + error.message; } }finally{ clearTimeout(timer); button.disabled = false; button.textContent = "✨ Generate Complete Video Production Kit"; } }
